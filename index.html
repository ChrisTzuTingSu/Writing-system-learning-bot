const registeredLanguages = [window.thaiData]; let currentLang = null; let currentIndex = 0; const canvas = document.getElementById('paintCanvas'); const ctx = canvas.getContext('2d'); let isDrawing = false; let passedCheckpoints = []; function showPage(pageId) { document.querySelectorAll('.page').forEach(p => p.classList.remove('active')); document.getElementById(pageId).classList.add('active'); } function initHome() { const langList = document.getElementById('langList'); langList.innerHTML = ''; registeredLanguages.forEach(lang => { if (!lang) return; const card = document.createElement('div'); card.className = 'card lang-card'; card.onclick = () => openOverview(lang); card.innerHTML = `
${lang.title}
收錄 ${lang.items.length} 個符號與聲調
→ `; langList.appendChild(card); }); } function openOverview(lang) { currentLang = lang; document.getElementById('overviewTitle').innerText = lang.title; document.getElementById('overviewDesc').innerText = lang.description; const grid = document.getElementById('overviewGrid'); grid.innerHTML = ''; lang.items.forEach((item, idx) => { const el = document.createElement('div'); el.className = 'grid-item'; el.onclick = () => playAudio(item.char, lang.langCode); el.innerHTML = ` ${item.className}
${item.char}
${item.name}
${item.ipa}｜${item.zhuyin}
`; grid.appendChild(el); }); showPage('overviewPage'); } function startPractice(index) { currentIndex = index; showPage('practicePage'); initCanvas(); loadPracticeItem(); } function initCanvas() { ctx.lineWidth = 10; ctx.lineCap = 'round'; ctx.lineJoin = 'round'; ctx.strokeStyle = '#1f2937'; } function drawCheckpointsOverlay() { const item = currentLang.items[currentIndex]; if (!item.checkpoints) return; item.checkpoints.forEach((pt, idx) => { const isPassed = passedCheckpoints.includes(idx); ctx.beginPath(); ctx.arc(pt.x, pt.y, 14, 0, Math.PI * 2); ctx.fillStyle = isPassed ? '#10b981' : '#e5e7eb'; ctx.fill(); ctx.strokeStyle = isPassed ? '#059669' : '#9ca3af'; ctx.lineWidth = 2; ctx.stroke(); ctx.fillStyle = isPassed ? '#ffffff' : '#374151'; ctx.font = 'bold 12px sans-serif'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle'; ctx.fillText(idx + 1, pt.x, pt.y); }); initCanvas(); } function loadPracticeItem() { resetCanvas(); const item = currentLang.items[currentIndex]; document.getElementById('practiceProgress').innerText = `${currentIndex + 1} / ${currentLang.items.length}`; document.getElementById('practiceName').innerText = item.name + " (" + item.meaning + ")"; document.getElementById('practiceBadge').innerText = item.className; document.getElementById('practiceBadge').className = `badge ${item.class}`; document.getElementById('practiceIpa').innerText = "音標與說明：" + item.ipa; document.getElementById('practiceZhuyin').innerText = "擬音與發音：" + item.zhuyin; document.getElementById('canvasGuide').innerText = item.char; passedCheckpoints = []; drawCheckpointsOverlay(); updateStatus("請由點位 1 依次劃過所有數字節點", ""); playCurrentAudio(); } function resetCanvas() { ctx.clearRect(0, 0, canvas.width, canvas.height); passedCheckpoints = []; drawCheckpointsOverlay(); updateStatus("請由點位 1 依次劃過所有數字節點", ""); } function getCanvasPos(e) { const rect = canvas.getBoundingClientRect(); const clientX = e.touches ? e.touches[0].clientX : e.clientX; const clientY = e.touches ? e.touches[0].clientY : e.clientY; return { x: clientX - rect.left, y: clientY - rect.top }; } function checkStrokeCheckpoints(pos) { const item = currentLang.items[currentIndex]; if (!item.checkpoints) return; const nextTargetIndex = passedCheckpoints.length; const targetPoint = item.checkpoints[nextTargetIndex]; if (targetPoint) { const dist = Math.hypot(pos.x - targetPoint.x, pos.y - targetPoint.y); // 判定容許半徑設為 35 像素 if (dist < 35) { passedCheckpoints.push(nextTargetIndex); drawCheckpointsOverlay(); } } } function startDrawing(e) { isDrawing = true; const pos = getCanvasPos(e); ctx.beginPath(); ctx.moveTo(pos.x, pos.y); checkStrokeCheckpoints(pos); } function draw(e) { if (!isDrawing) return; const pos = getCanvasPos(e); ctx.lineTo(pos.x, pos.y); ctx.stroke(); checkStrokeCheckpoints(pos); } function stopDrawing() { isDrawing = false; } canvas.addEventListener('mousedown', startDrawing); canvas.addEventListener('mousemove', draw); canvas.addEventListener('mouseup', stopDrawing); canvas.addEventListener('touchstart', (e) => { e.preventDefault(); startDrawing(e); }); canvas.addEventListener('touchmove', (e) => { e.preventDefault(); draw(e); }); canvas.addEventListener('touchend', stopDrawing); function nextPractice() { const item = currentLang.items[currentIndex]; const requiredCount = item.checkpoints ? item.checkpoints.length : 0; if (passedCheckpoints.length >= requiredCount) { updateStatus("筆順正確！進入下一個項目", "success"); setTimeout(() => { if (currentIndex < currentLang.items.length - 1) { currentIndex++; loadPracticeItem(); } else { alert("已完成本模組所有字母與聲調練習！"); showPage('overviewPage'); } }, 400); } else { updateStatus("未依序劃過所有點位，請重新書寫", "error"); } } function prevPractice() { if (currentIndex > 0) { currentIndex--; loadPracticeItem(); } } function updateStatus(msg, type) { const el = document.getElementById('statusMsg'); el.innerText = msg; el.className = `status-msg ${type}`; } function playAudio(char, langCode) { if ('speechSynthesis' in window) { window.speechSynthesis.cancel(); const utterance = new SpeechSynthesisUtterance(char); utterance.lang = langCode; utterance.rate = 0.8; window.speechSynthesis.speak(utterance); } } function playCurrentAudio() { const item = currentLang.items[currentIndex]; playAudio(item.char, currentLang.langCode); } initHome();
